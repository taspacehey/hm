// Google Apps Script - গোপন নোটের ব্যাকএন্ড // নোট: Google Sheet-এ (এনক্রিপ্টেড) | ছবি: Google Drive-এ (এনক্রিপ্টেড) const SHEET_NAME = 'Notes'; const FOLDER_NAME = 'GoponNoteFiles'; function getSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let s = ss.getSheetByName(SHEET_NAME); if (!s) { s = ss.insertSheet(SHEET_NAME); s.appendRow(['id', 'time', 'device', 'data']); } return s; } function folder_() { const it = DriveApp.getFoldersByName(FOLDER_NAME); return it.hasNext() ? it.next() : DriveApp.createFolder(FOLDER_NAME); } function isOurFile_(f) { const n = f.getName(); return n.indexOf('gn-') === 0 && n.slice(-4) === '.enc'; } function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { // ছবির এনক্রিপ্টেড ফাইল চাইলে if (e && e.parameter && e.parameter.file) { try { const f = DriveApp.getFileById(e.parameter.file); if (!isOurFile_(f)) return ContentService.createTextOutput(''); return ContentService.createTextOutput(f.getBlob().getDataAsString()); } catch (err) { return ContentService.createTextOutput(''); } } // নোটের তালিকা const rows = getSheet_().getDataRange().getValues().slice(1).map(function (r) { return { id: String(r[0]), time: Number(r[1]), device: String(r[2]), data: String(r[3]) }; }); return json_(rows); } function doPost(e) { const lock = LockService.getScriptLock(); lock.waitLock(30000); try { const b = JSON.parse(e.postData.contents); if (b.action === 'upload') { const name = 'gn-' + Date.now() + '-' + Math.random().toString(36).slice(2, 8) + '.enc'; const f = folder_().createFile(name, String(b.data), MimeType.PLAIN_TEXT); return json_({ ok: true, fid: f.getId() }); } const sheet = getSheet_(); if (b.action === 'add') { sheet.appendRow([String(b.id), Number(b.time), String(b.device), String(b.data)]); return json_({ ok: true }); } if (b.action === 'del') { const ids = sheet.getRange(1, 1, sheet.getLastRow(), 1).getValues(); for (let i = ids.length - 1; i >= 1; i--) { if (String(ids[i][0]) === String(b.id)) { sheet.deleteRow(i + 1); break; } } if (b.fid) { try { const f = DriveApp.getFileById(String(b.fid)); if (isOurFile_(f)) f.setTrashed(true); } catch (err) {} } return json_({ ok: true }); } return json_({ ok: false }); } finally { lock.releaseLock(); } }
